# nothing volatile: no random, time, os.environ, process.env or pid; no uncaught node throw,
# whose stack footer prints node's version; and no path, since /tmp/batch<N> moves with a case

# --- the language runs at all ---
[["print(6*7)","python"]]
[["console.log(6*7)","node"]]

# --- a string element is still SQL, and elements may be mixed ---
[["select 1 as sql_pair",""]]
[["select 1 as first",""],"select 2 as second",["print(3)","python"]]
["select 1 as plain",["print(2)","python"],"select 3 as plain_again"]

[["create table t as select 42 x",""],["import duckdb\nc = duckdb.connect('/data/fiddle.duckdb')\nprint(c.execute('select sum(x) from t').fetchall())","python"]]

# --- one process per batch: nothing carries over ---
